   181→function AnalysisPanel({ work, onSynopsis }: { work: WorkData; onSynopsis: (pt: string) => void }) {
   182→  const { toast } = useToast();
   183→  const [analysis, setAnalysis] = useState<string | null>(null);
   184→  const [loading, setLoading] = useState(false);
   185→  const [phase, setPhase] = useState(0);
   186→  const phases = ["Lendo os dados da obra...", "Cruzando tags e parecidas...", "Aoi escrevendo o veredito..."];
   187→
   188→  useEffect(() => {
   189→    if (!loading) return;
   190→    const t = setInterval(() => setPhase((p) => (p + 1) % phases.length), 1600);
   191→    return () => clearInterval(t);
   192→  }, [loading]);
   193→
   194→  const run = async () => {
   195→    setLoading(true);
   196→    setAnalysis(null);
   197→    try {
   198→      const res = await fetch("/api/ai/analyze", {
   199→        method: "POST",
   200→        headers: { "Content-Type": "application/json" },
   201→        body: JSON.stringify({ id: work.id }),
   202→      });
   203→      const d = await res.json();
   204→      if (!d.ok) throw new Error(d.error);
   205→      setAnalysis(d.analysis);
   206→      if (d.synopsisPT) onSynopsis(d.synopsisPT);
   207→    } catch (e: any) {
   208→      toast({ title: "A IA pisou na bola", description: e.message ?? "Tente de novo em instantes.", variant: "destructive" });
   209→    } finally {
   210→      setLoading(false);
   211→    }
   212→  };
   213→
   214→  if (!analysis && !loading) {
   215→    return (
   216→      <div className="flex flex-col items-center gap-4 rounded-2xl border border-dashed border-primary/30 bg-primary/5 p-10 text-center">
   217→        <span className="bg-brand-gradient flex h-14 w-14 items-center justify-center rounded-2xl text-2xl shadow-lg shadow-violet-950/40">
   218→          🍡
   219→        </span>
   220→        <div>
   221→          <p className="font-extrabold">Análise da curadora IA — Aoi</p>
   222→          <p className="mx-auto mt-1 max-w-md text-sm text-muted-foreground">
   223→            Ela cruza nota, tags, parecidas e histórico pra gerar um veredito honesto: arte, história, ritmo, nível de picância e pra quem é.
   224→          </p>
   225→        </div>
   226→        <Button onClick={run} className="bg-brand-gradient font-bold text-white hover:brightness-110">
   227→          <Sparkles className="h-4 w-4" /> Gerar análise em português
   228→        </Button>
   229→      </div>
   230→    );
   231→  }
   232→
   233→  if (loading) {
   234→    return (
   235→      <div className="flex flex-col items-center gap-4 rounded-2xl border border-primary/25 bg-card/60 p-10 text-center">
   236→        <Loader2 className="h-8 w-8 animate-spin text-primary" />
   237→        <p className="text-sm font-semibold text-violet-200">{phases[phase]}</p>
   238→        <p className="text-xs text-muted-foreground">a Aoi está caprichando — costuma levar ~15s</p>
   239→      </div>
   240→    );
   241→  }
   242→
   243→  return (
   244→    <div className="fade-in rounded-2xl border border-primary/25 bg-card/70 p-5 sm:p-7">
   245→      <div className="markdown-aoi">
   246→        <AoiMarkdown content={analysis ?? ""} />
   247→      </div>
   248→      <div className="mt-5 flex justify-end">
   249→        <Button variant="outline" size="sm" onClick={run} disabled={loading} className="border-primary/30 text-violet-200 hover:bg-primary/10">
   250→          <Sparkles className="h-3.5 w-3.5" /> Regenerar análise
   251→        </Button>
   252→      </div>
   253→    </div>
   254→  );
   255→}

   263→
   264→  useEffect(() => {
   265→    let alive = true;
   266→    setData(null);
   267→    setError(null);
   268→    setSynopsisPT(null);
